05

Accessibility & privacy

Contrast computed live for the current theme. WCAG AA throughout.

Contrast · live

Aa {{ p.label }}{{ p.note }} {{ p.ratio }}:1 {{ p.badge }}

Built-in for AA

  • Visible 2px focus ring on every interactive element, 3:1+ against its background.
  • Source & priority never rely on color alone — always paired icon + text.
  • Full keyboard path: ⌘K, arrow nav, E/S shortcuts, roving tabindex in the list.
  • Semantic roles: list/listitem rows, dialog palette, aria-live for sync & errors.
  • Respects prefers-reduced-motion and prefers-color-scheme.

Privacy-first by design

  • 01Remote images & trackers blocked by default — opt in per sender, never globally.
  • 02Search and command palette run on-device. No query telemetry.
  • 03No read receipts sent unless you turn them on. Previews can be redacted.
  • 04Data minimization: the UI asks for the least it needs, with clear retention & one-tap local wipe.

InboxItem.tsx
export function InboxItem({ msg, active, onOpen }: Props) {
  return (
    <button
      onClick={onOpen}
      aria-label={`${msg.sender}: ${msg.subject}`}
      data-active={active}
      className="group relative flex w-full gap-3 rounded-lg
        px-4 py-[15px] text-left
        hover:bg-surface-2
        focus-visible:outline-2 focus-visible:outline-accent
        focus-visible:outline-offset-2
        data-[active=true]:bg-surface-2"
    >
      <span className="absolute left-0 inset-y-3 w-[3px] rounded-r
        bg-transparent group-data-[active=true]:bg-accent" />
      <Avatar initials={msg.initials} source={msg.source} />
      <span className="min-w-0 flex-1 space-y-[3px]">
        <span className="flex items-baseline gap-2">
          <span className={cn("truncate text-sm text-text",
            msg.unread ? "font-bold" : "font-medium")}>
            {msg.sender}
          </span>
          <time className="ml-auto font-mono text-[11px]
            text-text-3">{msg.time}</time>
          {msg.unread && (
            <span className="size-[7px] rounded-full bg-accent" />
          )}
        </span>
        <SubjectLine msg={msg} />
        <Snippet msg={msg} />
      </span>
    </button>
  );
}
Badges.tsx
const SOURCE = {
  email:    { label: "Email",    Icon: Mail, tone: "slate" },
  slack:    { label: "Slack",    Icon: Hash, tone: "plum"  },
  telegram: { label: "Telegram", Icon: Send, tone: "steel" },
} as const;

export function SourceBadge({ source }: { source: Source }) {
  const { label, Icon, tone } = SOURCE[source];
  return (
    <span data-tone={tone}
      className="inline-flex items-center gap-1.5
        rounded-full px-2.5 py-1 text-xs
        data-[tone=slate]:bg-slate-soft data-[tone=slate]:text-slate
        data-[tone=plum]:bg-plum-soft  data-[tone=plum]:text-plum
        data-[tone=steel]:bg-steel-soft data-[tone=steel]:text-steel">
      <Icon className="size-3.5" aria-hidden /> {label}
    </span>
  );
}

// Only Urgent / High render a colored pill — calm by default.
export function PriorityBadge({ level }: { level: Priority }) {
  if (level === "normal" || level === "low") return null;
  const tone = level === "urgent" ? "danger" : "warning";
  return (
    <span data-tone={tone}
      className="inline-flex items-center gap-1.5 rounded-full
        px-2 py-0.5 font-mono text-[10.5px]
        data-[tone=danger]:bg-danger-soft  data-[tone=danger]:text-danger
        data-[tone=warning]:bg-warning-soft data-[tone=warning]:text-warning">
      <span className="size-[5px] rounded-full bg-current" />
      {level === "urgent" ? "Urgent" : "High"}
    </span>
  );
}
